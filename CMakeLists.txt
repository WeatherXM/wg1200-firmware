# The following lines of boilerplate have to be in your project's CMakeLists
# in this exact order for cmake to work correctly
cmake_minimum_required(VERSION 3.16)

# ------------------------------------------------------------------------------
# Resolve PROJECT_VER (used by ESP-IDF to populate esp_app_desc_t.version)
# Must be defined BEFORE including project.cmake
# ------------------------------------------------------------------------------

# 1. CI/CD or build command override via environment variable
if(DEFINED ENV{FIRMWARE_VERSION} AND NOT "$ENV{FIRMWARE_VERSION}" STREQUAL "")
    set(PROJECT_VER "$ENV{FIRMWARE_VERSION}")
# 2. version.txt generated by pre_actions.py (Single Source of Truth)
elseif(EXISTS "${CMAKE_CURRENT_LIST_DIR}/version.txt")
    file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/version.txt" VERSION_LINE)
    list(GET VERSION_LINE 0 FIRST_LINE)
    string(STRIP "${FIRST_LINE}" PROJECT_VER)
# 3. Direct fallback parse from main/include/app_config.h
elseif(EXISTS "${CMAKE_CURRENT_LIST_DIR}/main/include/app_config.h")
    file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/main/include/app_config.h" APP_VER_LINE REGEX "^#define[ \t]+APP_VERSION_BASE[ \t]+\"([^\"]+)\"")
    if(APP_VER_LINE MATCHES "^#define[ \t]+APP_VERSION_BASE[ \t]+\"([^\"]+)\"")
        set(PROJECT_VER "${CMAKE_MATCH_1}")
    endif()
endif()

# Fallback default if not resolved
if(NOT DEFINED PROJECT_VER OR "${PROJECT_VER}" STREQUAL "")
    set(PROJECT_VER "0.0.0")
endif()

message(STATUS "Firmware Project Version (esp_app_desc): ${PROJECT_VER}")

include($ENV{IDF_PATH}/tools/cmake/project.cmake)

project(wg1010_port)
